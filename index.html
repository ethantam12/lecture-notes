// ------------------------------------------------------------------ // Cloudflare Worker: secure relay between the lecture notes site and // the Gemini API. Keeps your Gemini API key private (never sent to // the browser) by holding it as a Worker secret instead. // // Deploy this on workers.cloudflare.com (free), then set a secret // named GEMINI_API_KEY with your key from Google AI Studio. // ------------------------------------------------------------------ const ALLOWED_ORIGIN = "https://ethantam12.github.io"; const MODEL = "gemini-3.5-flash-lite"; // fastest/cheapest current model — check https://ai.google.dev/gemini-api/docs/models if this ever errors as "not found" — model names change over time function corsHeaders() { return { "Access-Control-Allow-Origin": ALLOWED_ORIGIN, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; } function arrayBufferToBase64(buffer) { let binary = ""; const bytes = new Uint8Array(buffer); const chunkSize = 0x8000; for (let i = 0; i < bytes.length; i += chunkSize) { binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunkSize)); } return btoa(binary); } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders() }); } if (request.method !== "POST") { return new Response("Method not allowed", { status: 405, headers: corsHeaders() }); } let body; try { body = await request.json(); } catch (e) { return new Response(JSON.stringify({ error: "Invalid JSON body" }), { status: 400, headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } const question = String(body.question || "").slice(0, 4000); // Keep the END (most recent part) of the transcript, not the start — that's // what matters most for "what did they just say" questions during a live // recording, and a smaller prompt also means a noticeably faster answer. const fullTranscript = String(body.transcript || ""); const transcript = fullTranscript.length > 24000 ? fullTranscript.slice(-24000) : fullTranscript; const notes = String(body.notes || "").slice(0, 12000); const slideUrls = Array.isArray(body.slideUrls) ? body.slideUrls.slice(0, 8) : []; if (!question) { return new Response(JSON.stringify({ error: "Missing question" }), { status: 400, headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } let promptText = "You are a helpful study assistant embedded in a student's lecture notes app. " + "Answer the student's question using the lecture transcript, their notes, and any slide images/PDFs provided below as context. " + "Be concise and direct. If the context doesn't contain the answer, say so plainly rather than guessing.\n\n"; if (transcript) promptText += "LECTURE TRANSCRIPT SO FAR:\n" + transcript + "\n\n"; if (notes) promptText += "STUDENT'S NOTES:\n" + notes + "\n\n"; promptText += "QUESTION: " + question; const parts = [{ text: promptText }]; // Fetch slide images/PDFs in parallel instead of one-at-a-time — with // several slides attached this alone used to add seconds of latency. const slideParts = await Promise.all(slideUrls.map(async (url) => { try { const res = await fetch(url); if (!res.ok) return null; const contentType = res.headers.get("content-type") || "application/octet-stream"; if (!/^image\/|^application\/pdf/.test(contentType)) return null; const buf = await res.arrayBuffer(); if (buf.byteLength > 15 * 1024 * 1024) return null; // skip anything oversized return { inline_data: { mime_type: contentType, data: arrayBufferToBase64(buf) } }; } catch (e) { return null; // skip slides that fail to fetch } })); slideParts.forEach((p) => { if (p) parts.push(p); }); try { const geminiRes = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${env.GEMINI_API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts }], generationConfig: { // Skip/minimize the model's internal "thinking" step — this is // by far the biggest lever for making answers come back faster. thinkingConfig: { thinkingLevel: "minimal" }, maxOutputTokens: 700 } }) } ); const data = await geminiRes.json(); if (!geminiRes.ok) { const msg = (data && data.error && data.error.message) || "Gemini request failed"; return new Response(JSON.stringify({ error: msg }), { status: 502, headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } const candidate = data.candidates && data.candidates[0]; const answer = (candidate && candidate.content && candidate.content.parts && candidate.content.parts.map(p => p.text || "").join("")) || "No answer returned."; return new Response(JSON.stringify({ answer }), { headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } catch (e) { return new Response(JSON.stringify({ error: e.message || "Unexpected error" }), { status: 500, headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } } };
