// ------------------------------------------------------------------ // Cloudflare Worker: secure relay between the lecture notes site and // the Gemini API. Keeps your Gemini API key private (never sent to // the browser) by holding it as a Worker secret instead. // // Deploy this on workers.cloudflare.com (free), then set a secret // named GEMINI_API_KEY with your key from Google AI Studio. // // Supports four request modes (body.mode): // "ask" (default) — one-shot question + answer, used by the // per-lecture and per-course "Ask AI" panels. // "chat" — multi-turn tutor conversation (body.history is the // prior back-and-forth), used by the Tutor Chat panel. // "generate" — structured JSON generation (body.genType is // "flashcards" or "quiz"), used by the Flashcards/Quiz // features. // "transcribe" — sends lecture audio (either body.audioUrl for a // saved recording, or body.audioBase64 for a short // live chunk) to Gemini for a full, accurate transcript // — much more complete than the browser's live captions. // ------------------------------------------------------------------ const ALLOWED_ORIGIN = "https://ethantam12.github.io"; const MODEL = "gemini-3.5-flash-lite"; // fastest/cheapest current model — check https://ai.google.dev/gemini-api/docs/models if this ever errors as "not found" — model names change over time function corsHeaders() { return { "Access-Control-Allow-Origin": ALLOWED_ORIGIN, "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; } function jsonResponse(obj, status) { return new Response(JSON.stringify(obj), { status: status || 200, headers: { ...corsHeaders(), "Content-Type": "application/json" } }); } function arrayBufferToBase64(buffer) { let binary = ""; const bytes = new Uint8Array(buffer); const chunkSize = 0x8000; for (let i = 0; i < bytes.length; i += chunkSize) { binary += String.fromCharCode.apply(null, bytes.subarray(i, i + chunkSize)); } return btoa(binary); } function httpError(message, status) { return Object.assign(new Error(message), { status: status }); } // Fetch slide images/PDFs in parallel instead of one-at-a-time — with // several slides attached this alone used to add seconds of latency. async function fetchSlideParts(slideUrls) { const results = await Promise.all(slideUrls.map(async (url) => { try { const res = await fetch(url); if (!res.ok) return null; const contentType = res.headers.get("content-type") || "application/octet-stream"; if (!/^image\/|^application\/pdf/.test(contentType)) return null; const buf = await res.arrayBuffer(); if (buf.byteLength > 15 * 1024 * 1024) return null; // skip anything oversized return { inline_data: { mime_type: contentType, data: arrayBufferToBase64(buf) } }; } catch (e) { return null; // skip slides that fail to fetch } })); return results.filter(Boolean); } // Downloads the recorded lecture audio and prepares it as an inline part for // Gemini. Kept well under Gemini's 20MB inline-request limit so there's room // left for the prompt text itself. async function fetchAudioPart(audioUrl) { let res; try { res = await fetch(audioUrl); } catch (e) { throw httpError("Couldn't download the recording's audio.", 502); } if (!res.ok) { throw httpError("Couldn't download the recording's audio.", 502); } const contentType = (res.headers.get("content-type") || "audio/webm").split(";")[0]; const buf = await res.arrayBuffer(); if (buf.byteLength > 19 * 1024 * 1024) { throw httpError("This recording is too long for AI transcription right now (roughly 90+ minutes at your recording quality). The live transcript captured while recording is still available.", 413); } return { inline_data: { mime_type: contentType, data: arrayBufferToBase64(buf) } }; } function buildContextText(transcript, notes) { let ctx = ""; if (transcript) ctx += "LECTURE TRANSCRIPT:\n" + transcript + "\n\n"; if (notes) ctx += "STUDENT'S NOTES:\n" + notes + "\n\n"; return ctx; } async function callGemini(env, requestBody) { const res = await fetch( `https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent?key=${env.GEMINI_API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(requestBody) } ); const data = await res.json(); if (!res.ok) { const msg = (data && data.error && data.error.message) || "Gemini request failed"; throw httpError(msg, 502); } const candidate = data.candidates && data.candidates[0]; const text = (candidate && candidate.content && candidate.content.parts && candidate.content.parts.map(p => p.text || "").join("")) || ""; return text; } function flashcardSchema() { return { type: "ARRAY", items: { type: "OBJECT", properties: { front: { type: "STRING", description: "The question or term side of the flashcard." }, back: { type: "STRING", description: "The answer or definition side of the flashcard." } }, required: ["front", "back"] } }; } function quizSchema() { return { type: "ARRAY", items: { type: "OBJECT", properties: { question: { type: "STRING" }, choices: { type: "ARRAY", items: { type: "STRING" } }, correctIndex: { type: "INTEGER", description: "Index (0-3) into choices of the correct answer." }, explanation: { type: "STRING", description: "One or two sentences on why that answer is correct." } }, required: ["question", "choices", "correctIndex", "explanation"] } }; } export default { async fetch(request, env) { if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders() }); } if (request.method !== "POST") { return new Response("Method not allowed", { status: 405, headers: corsHeaders() }); } let body; try { body = await request.json(); } catch (e) { return jsonResponse({ error: "Invalid JSON body" }, 400); } const mode = String(body.mode || "ask"); // Keep the END (most recent part) of the transcript, not the start — that's // what matters most for "what did they just say" questions during a live // recording, and a smaller prompt also means a noticeably faster answer. const fullTranscript = String(body.transcript || ""); const transcript = fullTranscript.length > 24000 ? fullTranscript.slice(-24000) : fullTranscript; const notes = String(body.notes || "").slice(0, 12000); const slideUrls = Array.isArray(body.slideUrls) ? body.slideUrls.slice(0, 8) : []; try { if (mode === "chat") { const question = String(body.question || "").slice(0, 4000); if (!question) return jsonResponse({ error: "Missing question" }, 400); const history = Array.isArray(body.history) ? body.history.slice(-20) : []; const slideParts = await fetchSlideParts(slideUrls); const systemText = "You are Sparky, a friendly, encouraging AI study tutor embedded in a student's lecture notes app. " + "Have a natural back-and-forth conversation and reference the lecture transcript/notes below when relevant. " + "Keep answers focused; expand only if the student asks for more depth. " + "If the context doesn't cover something, say so plainly rather than guessing.\n\n" + buildContextText(transcript, notes); const contents = history.map((h) => ({ role: h.role === "assistant" ? "model" : "user", parts: [{ text: String(h.text || "").slice(0, 4000) }] })); const lastParts = [{ text: question }]; slideParts.forEach((p) => lastParts.push(p)); contents.push({ role: "user", parts: lastParts }); const answer = await callGemini(env, { systemInstruction: { parts: [{ text: systemText }] }, contents: contents, generationConfig: { thinkingConfig: { thinkingLevel: "minimal" }, maxOutputTokens: 700 } }); return jsonResponse({ answer: answer || "No answer returned." }); } if (mode === "transcribe") { // Either a full recording fetched from its storage URL (the "Improve // with AI" button, after saving) or a short raw audio chunk sent // straight from the browser as base64 (live in-progress recording, // re-transcribed every ~30s for far better accuracy than the // browser's built-in live captions). let audioPart; if (body.audioBase64) { const b64 = String(body.audioBase64); if (b64.length > 26 * 1024 * 1024) return jsonResponse({ error: "Audio chunk too large" }, 413); const mimeType = String(body.mimeType || "audio/webm").split(";")[0]; audioPart = { inline_data: { mime_type: mimeType, data: b64 } }; } else { const audioUrl = String(body.audioUrl || ""); if (!audioUrl) return jsonResponse({ error: "Missing audioUrl or audioBase64" }, 400); audioPart = await fetchAudioPart(audioUrl); } const promptText = "Transcribe this lecture audio verbatim, as accurately and completely as you can. " + "It may be a short excerpt from a longer recording — transcribe only what is actually heard, " + "with no introduction, commentary, or note about it being an excerpt. " + "Capture everything that is said — do not summarize, paraphrase, or skip parts. " + "If a portion is inaudible or unclear, write [inaudible] rather than guessing at it."; const newTranscript = await callGemini(env, { contents: [{ parts: [{ text: promptText }, audioPart] }], generationConfig: { thinkingConfig: { thinkingLevel: "minimal" }, maxOutputTokens: 8192 } }); if (!newTranscript) return jsonResponse({ error: "The AI didn't return a transcript. Try again." }, 502); return jsonResponse({ transcript: newTranscript }); } if (mode === "generate") { const genType = body.genType === "quiz" ? "quiz" : "flashcards"; const count = Math.max(3, Math.min(20, parseInt(body.count, 10) || 10)); if (!transcript && !notes) return jsonResponse({ error: "No transcript or notes to generate from" }, 400); const promptText = (genType === "quiz" ? "Write exactly " + count + " multiple-choice quiz questions (4 choices each) testing understanding of the material below. " + "Vary difficulty and cover different parts of the material. Each question needs exactly 4 choices, a correctIndex (0-3), and a short explanation." : "Write exactly " + count + " study flashcards (front = question/term, back = concise answer/definition) covering the key concepts in the material below. " + "Favor the most exam-relevant ideas over trivia.") + "\n\n" + buildContextText(transcript, notes); const raw = await callGemini(env, { contents: [{ parts: [{ text: promptText }] }], generationConfig: { thinkingConfig: { thinkingLevel: "low" }, maxOutputTokens: 4000, responseMimeType: "application/json", responseSchema: genType === "quiz" ? quizSchema() : flashcardSchema() } }); let items; try { items = JSON.parse(raw); } catch (e) { items = null; } if (!Array.isArray(items)) return jsonResponse({ error: "The AI didn't return usable results. Try again." }, 502); return jsonResponse({ items: items }); } // mode === "ask" (default) — one-shot question, unchanged behavior. const question = String(body.question || "").slice(0, 4000); if (!question) return jsonResponse({ error: "Missing question" }, 400); const slideParts = await fetchSlideParts(slideUrls); let promptText = "You are a helpful study assistant embedded in a student's lecture notes app. " + "Answer the student's question using the lecture transcript, their notes, and any slide images/PDFs provided below as context. " + "Be concise and direct. If the context doesn't contain the answer, say so plainly rather than guessing.\n\n"; promptText += buildContextText(transcript, notes); promptText += "QUESTION: " + question; const parts = [{ text: promptText }]; slideParts.forEach((p) => parts.push(p)); const answer = await callGemini(env, { contents: [{ parts }], generationConfig: { // Skip/minimize the model's internal "thinking" step — this is // by far the biggest lever for making answers come back faster. thinkingConfig: { thinkingLevel: "minimal" }, maxOutputTokens: 700 } }); return jsonResponse({ answer: answer || "No answer returned." }); } catch (e) { return jsonResponse({ error: e.message || "Unexpected error" }, e.status || 500); } } };
